# The PGRAPH core and GL renderer both come from xemu at 478b4f4.
set(nv "${NGB_ROOT}/xboxrecomp/src/nv2a_vk")
set(tp "${NGB_ROOT}/xboxrecomp/third_party")
set(pg "${nv}/xemu/pgraph")
file(GLOB sources CONFIGURE_DEPENDS "${pg}/*.c" "${pg}/gl/*.c" "${pg}/glsl/*.c" "${tp}/nv2a_vsh_cpu/src/*.c")
add_library(ngb_scene STATIC bridge.c context.c present.c ${sources} "${nv}/shim/util/fast-hash.c" "${nv}/shim/glib-lite/glib_lite.c")
target_compile_definitions(ngb_scene PRIVATE NGB_SWITCH_PGRAPH CONFIG_OPENGL XXH_INLINE_ALL)
target_compile_options(ngb_scene PRIVATE -O2 -ffunction-sections -fdata-sections -fno-strict-aliasing -fvisibility=hidden)
target_include_directories(ngb_scene SYSTEM PRIVATE "${NGB_ROOT}/xboxrecomp/src")
target_include_directories(ngb_scene PRIVATE "." ".." "${pg}/gl" "${nv}/shim/glib-lite" "${nv}/shim" "${nv}/xemu" "${nv}/host"
    "${tp}/nv2a_vsh_cpu/src" "${tp}/xxHash" "${tp}/stb")
target_link_libraries(ngb_scene PUBLIC PkgConfig::MENU_GL)
if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    find_library(glad_library glad REQUIRED)
    target_link_libraries(ngb_scene PUBLIC "${glad_library}")
else()
    pkg_check_modules(OPENGL REQUIRED IMPORTED_TARGET gl)
    target_link_libraries(ngb_scene PUBLIC PkgConfig::OPENGL)
endif()
