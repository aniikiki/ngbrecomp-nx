include(FetchContent)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "")
set(ENABLE_HLSL ON CACHE BOOL "")
set(ENABLE_OPT OFF CACHE BOOL "")
set(ENABLE_PCH OFF CACHE BOOL "")
set(GLSLANG_TESTS OFF CACHE BOOL "")
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "")
FetchContent_Declare(glslang
    URL https://codeload.github.com/KhronosGroup/glslang/tar.gz/refs/tags/16.0.0
    URL_HASH SHA256=172385478520335147d3b03a1587424af0935398184095f24beab128a254ecc7)
FetchContent_MakeAvailable(glslang)
set(tp "${NGB_ROOT}/xboxrecomp/third_party")
set(nv "${NGB_ROOT}/xboxrecomp/src/nv2a_vk")
set(d3d "${NGB_ROOT}/src/d3d")
set(volk "${tp}/volk/volk.c")
if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    set(NGB_DIRECT_SPIRV "" CACHE FILEPATH "Optional host-generated SPIR-V cache to embed in RomFS")
    set(seed "${NGB_ROOT}/build/switch-runtime/program-romfs/.ngb-spirv")
    if(NGB_DIRECT_SPIRV)
        configure_file("${NGB_DIRECT_SPIRV}" "${seed}" COPYONLY)
        set_property(TARGET ngb-startup APPEND PROPERTY LINK_DEPENDS "${seed}")
    else()
        file(REMOVE "${seed}")
    endif()
    set(NGB_DIRECT_RECIPES "" CACHE FILEPATH "Optional Vulkan pipeline recipes to embed in RomFS")
    set(recipe_seed "${NGB_ROOT}/build/switch-runtime/program-romfs/.ngb-recipes")
    if(NGB_DIRECT_RECIPES)
        configure_file("${NGB_DIRECT_RECIPES}" "${recipe_seed}" COPYONLY)
        set_property(TARGET ngb-startup APPEND PROPERTY LINK_DEPENDS "${recipe_seed}")
    else()
        file(REMOVE "${recipe_seed}")
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(volk "${CMAKE_CURRENT_BINARY_DIR}/volk.c")
    add_custom_command(OUTPUT "${volk}"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/volk_switch.py" "${tp}/volk/volk.c" "${volk}"
        DEPENDS volk_switch.py "${tp}/volk/volk.c" VERBATIM)
    set(NGB_NXVK_ROOT "${NGB_ROOT}/../nxvk" CACHE PATH "Built nxvk checkout")
    set(cross "${NGB_NXVK_ROOT}/switch/build/cross")
    file(GLOB_RECURSE nvk_archives "${cross}/*.a")
    set(driver "${CMAKE_CURRENT_BINARY_DIR}/nvk.o")
    add_custom_command(OUTPUT "${driver}" COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/link-nvk.sh"
        "${NGB_NXVK_ROOT}" "${driver}" DEPENDS link-nvk.sh "${NGB_NXVK_ROOT}/switch/smoke/nvk_compat.c" ${nvk_archives} VERBATIM)
endif()
add_library(ngb_direct STATIC hooks.c platform.cpp "${d3d}/d3d_hooks.c" "${d3d}/hle.cpp"
    "${d3d}/gpu/threaded.cpp" "${d3d}/gpu/shaders.cpp" "${d3d}/gpu/texture.cpp" "${d3d}/gpu/vertex.cpp"
    "${d3d}/vk/backend.cpp" "${d3d}/vk/spirv.cpp" "${volk}" "${nv}/shim/util/vma.cc" ${driver})
target_compile_features(ngb_direct PRIVATE cxx_std_20)
target_compile_options(ngb_direct PRIVATE -O2 -ffunction-sections -fdata-sections -fno-strict-aliasing)
target_compile_definitions(ngb_direct PRIVATE NGB_DIRECT3D VK_NO_PROTOTYPES VMA_STATIC_VULKAN_FUNCTIONS=0 VMA_DYNAMIC_VULKAN_FUNCTIONS=0)
target_include_directories(ngb_direct PRIVATE . .. "${NGB_GEN}" "${NGB_ROOT}/src" "${d3d}" "${d3d}/compat"
    "${NGB_ROOT}/xboxrecomp/src" "${nv}/host" "${tp}/volk" "${tp}/Vulkan-Headers/include" "${tp}/VulkanMemoryAllocator/include")
target_link_libraries(ngb_direct PRIVATE glslang glslang-default-resource-limits ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    target_include_directories(ngb_direct PRIVATE "${NX_ROOT}/include")
    target_link_libraries(ngb_direct PRIVATE expat z nx)
endif()
