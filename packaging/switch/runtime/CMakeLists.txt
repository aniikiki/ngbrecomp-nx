cmake_minimum_required(VERSION 3.20)
project(ngb_startup C)
get_filename_component(NGB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(NGB_GEN "${NGB_ROOT}/build/switch-runtime/gen")
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${NGB_GEN}/recomp_*.c")
if(NOT GAME_SOURCES)
    message(FATAL_ERROR "Generate and stage the Switch boot code first")
endif()
option(NGB_DIRECT3D "Render the original Xbox Direct3D calls on Vulkan" OFF)
add_library(ngb_boot_code OBJECT ${GAME_SOURCES})
target_include_directories(ngb_boot_code PRIVATE "${NGB_GEN}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(ngb_boot_code PRIVATE -O2 -ffp-contract=off -fno-strict-aliasing -fwrapv
    -Wno-unused-label -Wno-unused-variable -Wno-unused-but-set-variable)
add_library(ngb_identity OBJECT "${NGB_ROOT}/xboxrecomp/src/kernel/kernel_xbox.c"
    "${NGB_ROOT}/xboxrecomp/src/kernel/kernel_rtl.c"
    "${NGB_ROOT}/xboxrecomp/src/kernel/kernel_crypto.c")
target_include_directories(ngb_identity PRIVATE "${NGB_ROOT}/xboxrecomp/src")
target_compile_options(ngb_identity PRIVATE -ffunction-sections -fdata-sections)
set_source_files_properties(scanout.c opening_movie.c menu_gpu.c pgraph_boot.c PROPERTIES COMPILE_OPTIONS "-O3")
add_executable(ngb-startup main.c runtime.c identity.c nv2a_boot.c scanout.c opening_movie.c menu_gpu.c pgraph_boot.c ohci_boot.c audio_boot.c contiguous.c vm.c timers.c files.c files_bridge.c $<TARGET_OBJECTS:ngb_boot_code> $<TARGET_OBJECTS:ngb_identity>)
target_include_directories(ngb-startup SYSTEM PRIVATE "${NGB_ROOT}/xboxrecomp/src")
target_link_options(ngb-startup PRIVATE -Wl,--gc-sections)
target_include_directories(ngb-startup PRIVATE "${NGB_GEN}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(ngb-startup PRIVATE -O1 -Wall -Wextra -Werror)
target_link_libraries(ngb-startup PRIVATE m)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MENU_GL REQUIRED IMPORTED_TARGET glesv2 egl)
add_subdirectory(scene)
pkg_check_modules(MOVIE REQUIRED IMPORTED_TARGET libavcodec libswscale libavutil)
target_link_libraries(ngb-startup PRIVATE ngb_scene PkgConfig::MOVIE PkgConfig::MENU_GL)
set(gamepad_control "${NGB_ROOT}/xboxrecomp/src/usb/usb_gamepad.c")
set_source_files_properties("${gamepad_control}" PROPERTIES COMPILE_DEFINITIONS XBOX_USB_EXTERNAL_REPORT)
target_sources(ngb-startup PRIVATE "${gamepad_control}")
if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    enable_language(CXX)
    set_property(TARGET ngb-startup PROPERTY LINKER_LANGUAGE CXX)
    target_link_libraries(ngb-startup PRIVATE nx)
    target_include_directories(ngb-startup PRIVATE "${NX_ROOT}/include")
    target_link_directories(ngb-startup PRIVATE "${NX_ROOT}/lib")
    file(MAKE_DIRECTORY "${NGB_ROOT}/build/switch-runtime/program-romfs")
    add_custom_command(TARGET ngb-startup POST_BUILD
        COMMAND "${NX_NACPTOOL_EXE}" --create "NGB Startup" "NGB port development" "0.49.0" ngb-startup.nacp
        COMMAND "${NX_ELF2NRO_EXE}" $<TARGET_FILE:ngb-startup> ngb-startup.nro
            --nacp=ngb-startup.nacp --icon=${NX_DEFAULT_ICON}
            --romfsdir=${NGB_ROOT}/build/switch-runtime/program-romfs
        VERBATIM)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    find_package(Threads REQUIRED)
    target_link_libraries(ngb-startup PRIVATE Threads::Threads)
    enable_testing()
    add_executable(ngb-vm-test vm_test.c vm.c)
    target_compile_options(ngb-vm-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME guest-virtual-memory COMMAND ngb-vm-test)
    add_executable(ngb-timers-test timers_test.c timers.c)
    target_compile_options(ngb-timers-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME guest-timers COMMAND ngb-timers-test)
    add_executable(ngb-files-test files_test.c files.c)
    target_compile_options(ngb-files-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME guest-files COMMAND ngb-files-test)
    add_executable(ngb-runtime-test runtime_test.c identity.c nv2a_boot.c scanout.c opening_movie.c menu_gpu.c pgraph_boot.c ohci_boot.c audio_boot.c contiguous.c vm.c timers.c files.c files_bridge.c
        $<TARGET_OBJECTS:ngb_boot_code> $<TARGET_OBJECTS:ngb_identity>)
    target_sources(ngb-runtime-test PRIVATE "${gamepad_control}")
    target_include_directories(ngb-runtime-test PRIVATE "${NGB_GEN}" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_include_directories(ngb-runtime-test SYSTEM PRIVATE "${NGB_ROOT}/xboxrecomp/src")
    target_compile_options(ngb-runtime-test PRIVATE -O1 -Wall -Wextra -Werror)
    target_link_options(ngb-runtime-test PRIVATE -Wl,--gc-sections -Wl,--wrap=ngb_nv2a_write32 -Wl,--wrap=ngb_kernel_log -Wl,--wrap=ngb_scanout_refresh -Wl,--wrap=ngb_scanout_publish_movie)
    target_link_libraries(ngb-runtime-test PRIVATE Threads::Threads ngb_scene m PkgConfig::MOVIE PkgConfig::MENU_GL)
    add_test(NAME guest-runtime-bridge COMMAND ngb-runtime-test "${NGB_ROOT}/game_files/default.xbe")
endif()

if(NGB_DIRECT3D)
    enable_language(CXX)
    include("${NGB_GEN}/direct_units.cmake")
    foreach(unit IN LISTS NGB_DIRECT_UNITS)
        set_property(SOURCE "${NGB_GEN}/${unit}" APPEND PROPERTY COMPILE_DEFINITIONS NGB_DIRECT3D)
    endforeach()
    add_subdirectory(direct)
    target_compile_definitions(ngb_scene PRIVATE NGB_DIRECT3D)
    foreach(app ngb-startup ngb-runtime-test)
        if(TARGET ${app})
            set_property(TARGET ${app} PROPERTY LINKER_LANGUAGE CXX)
            target_compile_definitions(${app} PRIVATE NGB_DIRECT3D)
            target_link_libraries(${app} PRIVATE ngb_direct)
        endif()
    endforeach()
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    # The guest library and kernel bridge must agree on the register file's ABI.
    target_include_directories(ngb_identity PRIVATE "${NGB_GEN}")
    foreach(target ngb_boot_code ngb_identity ngb_direct)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-include${CMAKE_CURRENT_SOURCE_DIR}/switch_guest_regs.h>")
        endif()
    endforeach()
endif()
